Por que precisamos de um back-end?

07

Imagine um sistema escolar

01 · Revisão

Arrays, funções e decisões

11

nomealunos.js

const alunos = [
    { nome: "Maria", nota: 8.5 },
    { nome: "Pedro", nota: 5.0 },
    { nome: "Ana", nota: 9.0 },
    { nome: "João", nota: 6.0 }
];

function verificarSituacao(nota) {
    if (nota >= 7) {
        return "Aprovado";
    }

    return "Reprovado";
}

for (let i = 0; i < alunos.length; i++) {
    const aluno = alunos[i];

    console.log(
        aluno.nome + " - " +
        aluno.nota + " - " +
        verificarSituacao(aluno.nota)
    );
}

Execução no terminal

node nomealunos.js
Maria - 8.5 - Aprovado
Pedro - 5 - Reprovado
Ana - 9 - Aprovado
João - 6 - Reprovado

O programa percorre o array e aplica uma regra para cada aluno. A regra não depende de uma interface gráfica.

02 · Exemplo 2

Receber notas e calcular a média

16

exemplo-prompt/avaliacao.js

const prompt = require("prompt-sync")();
const nome = prompt("Nome do aluno: ");
const nota1 = Number(prompt("Primeira nota: "));
const nota2 = Number(prompt("Segunda nota: "));
const media = (nota1 + nota2) / 2;

console.log("Aluno: " + nome);
console.log("Média: " + media);
if (media >= 7) {
    console.log("Aprovado");
} else {
    console.log("Reprovado");
}

Execução no terminal

node exemplo-prompt/avaliacao.js
Nome do aluno: Maria
Primeira nota: 8
Segunda nota: 9
Aluno: Maria
Média: 8.5
Aprovado

Number converte o texto em número. Neste exemplo, considere entradas válidas e use ponto para notas decimais.

02 · Exemplo 4

Um sistema com menu interativo

18

A ideia

Manter um programa em execução enquanto o usuário escolhe operações.

=== SISTEMA ESCOLAR ===
1 - Cadastrar aluno
2 - Listar alunos
3 - Sair
Escolha uma opção:

Conhecimentos reaproveitados

  • while para repetir o menu.
  • if / else if para escolher uma operação.
  • push para cadastrar.
  • for para listar.
Os dados ficam na memória durante a execução. Ao encerrar o programa, o array é perdido.
02 · Exemplo 4

Implementação do menu

19

sistema.js · início

const prompt = require("prompt-sync")();
const alunos = [];
let opcao = 0;

while (opcao !== 3) {
    console.log("\n=== SISTEMA ESCOLAR ===");
    console.log("1 - Cadastrar aluno");
    console.log("2 - Listar alunos");
    console.log("3 - Sair");
    opcao = Number(prompt("Escolha uma opção: "));

sistema.js · continuação

if (opcao === 1) {
        const nome = prompt("Nome do aluno: ");
        alunos.push(nome);
        console.log("Aluno cadastrado!");
    } else if (opcao === 2) {
        for (let i = 0; i < alunos.length; i++) {
            console.log(alunos[i]);
        }
    } else if (opcao === 3) {
        console.log("Encerrando...");
    } else {
        console.log("Opção inválida!");
    }
}
Os dois blocos fazem parte do mesmo arquivo. Execute: node exemplo-prompt/sistema.js.

03 · Transição

Do terminal ao servidor

21

Programa interativo

O usuário digita dados diretamente no terminal.

O programa processa e usa console.log() para apresentar resultados.

Servidor web

O navegador envia uma requisição pela rede.

O servidor processa e envia uma resposta HTTP.

A diferença principal é a forma de comunicação. O servidor permanece aguardando novas requisições.
03 · Primeiro servidor

Criando um servidor HTTP

22

servidor.js

const http = require("node:http");

const servidor = http.createServer(
    function (requisicao, resposta) {
        resposta.end("Meu primeiro servidor!");
    }
);

servidor.listen(3000);

Execução no terminal

node servidor.js

O texto entre colchetes é ilustrativo: o código não imprime mensagem no terminal. Abra http://localhost:3000 no navegador.

03 · HTTP

Entendendo o código

23

Criar o servidor

  • require("node:http"): carrega o módulo HTTP nativo.
  • createServer(): cria o servidor e recebe uma função chamada a cada requisição.
  • requisicao: informações do pedido.
  • resposta: objeto usado para responder.

Enviar e aguardar

  • resposta.end(): envia o conteúdo e finaliza a resposta.
  • listen(3000): inicia a escuta na porta 3000.
  • localhost: endereço do próprio computador.
O módulo node:http já acompanha o Node.js. Não é necessário instalá-lo com npm.

03 · Evolução

Enviando HTML na resposta

25

servidor-html.js

const http = require("node:http");

const servidor = http.createServer(
    function (requisicao, resposta) {
        resposta.setHeader("Content-Type", "text/html; charset=utf-8");

        resposta.end(`
            <h1>Meu primeiro servidor!</h1>
            <p>Esta página foi enviada pelo Node.js.</p>
            <hr>
            <h2>Disciplinas</h2>
            <ul>
                <li>Programação Web</li>
                <li>Banco de Dados</li>
                <li>Redes de Computadores</li>
            </ul>
        `);
    }
);

servidor.listen(3000);

Resultado no navegador

↻   localhost:3000
Meu primeiro servidor!

Esta página foi enviada pelo Node.js.

Disciplinas
  • Programação Web
  • Banco de Dados
  • Redes de Computadores

O navegador interpreta as tags e renderiza a página.

03 · Evolução

O que mudou?

26

Tipo de conteúdo

resposta.setHeader("Content-Type", "text/html; charset=utf-8");

Informa ao navegador que a resposta contém HTML codificado em UTF-8.

Texto em várias linhas

resposta.end(`
    <h1>Olá!</h1>
    <p>Minha página.</p>
`);

As crases permitem escrever uma string em várias linhas. O Node.js envia o texto; o navegador interpreta o HTML.

Encerre o servidor anterior com Ctrl + C antes de executar node servidor-html.js, pois ambos usam a porta 3000.
